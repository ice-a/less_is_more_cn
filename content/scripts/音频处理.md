## 音频格式转换

### Shell 方式（单文件）

```bash
# m4a 转 16bit、单声道、48kHz wav
for i in *.m4a; do ffmpeg -i "$i" -acodec pcm_s16le -ac 1 -ar 48000 "${i%}.wav"; done

# 参数说明
# -acodec pcm_s16le: 16bit
# -ac 1: 单声道
# -ar 48000: 48kHz
```

### Python 方式（批量，pydub 优先、失败回退 ffmpeg）

```python
import os
from pathlib import Path

from pydub import AudioSegment

def convert_to_wav(src_path, dst_path, audio_format):
    sound = AudioSegment.from_file(src_path, format=audio_format)
    sound = sound.set_frame_rate(16000).set_channels(1)
    sound.export(dst_path, format="wav")

def cmd_convert_to_wav(src_path, dst_path):
    os.system(f'ffmpeg -y -i "{src_path}" -ar 16000 -ac 1 -f wav "{dst_path}"')

def batch_convert(directory, filter_suffix=None, convert_suffix=None):
    filter_suffix = filter_suffix or []
    convert_suffix = convert_suffix or [".m4a", ".mp3", ".wav"]
    directory = Path(directory)
    for file in directory.iterdir():
        if not file.is_file():
            continue
        suffix = file.suffix.lower()
        if suffix in filter_suffix or suffix not in convert_suffix:
            continue
        wav_path = file.with_suffix(".wav")
        print(f"正在处理: {file}")
        try:
            convert_to_wav(str(file), str(wav_path), audio_format=suffix.lstrip("."))
        except Exception as e:
            print(f"pydub转换失败: {e}，尝试ffmpeg...")
            cmd_convert_to_wav(str(file), str(wav_path))

if __name__ == "__main__":
    batch_convert(r"音频目录路径")
```

## 音频时长统计

```python
import os

from pydub.utils import mediainfo

def get_audio_duration(file_path):
    return float(mediainfo(file_path)["duration"])

def count_folder_duration(path, file_types=None):
    file_types = [t.lower() for t in (file_types or ["wav", "mp3"])]
    total = 0
    for root, _, files in os.walk(path):
        for file in files:
            if file.rsplit(".", 1)[-1].lower() not in file_types:
                continue
            file_path = os.path.join(root, file)
            duration = get_audio_duration(file_path)
            print(f"{file_path}: {duration:.2f}秒")
            total += duration
    return total

if __name__ == "__main__":
    total = count_folder_duration(r"音频目录路径", ["wav", "mp3", "m4a"])
    print(f"总时长: {total:.2f}秒")
```

## 基于 VAD 的音频切割

场景：AI 生成的音频超过平台 60s 上限，按停顿切分后分段上传。原实现里 `dynamic_param_tuning` 算出的参数从未使用、`output_dir` 未创建、导出名 `segmentes_` 拼写错误，这里给精简可跑版本：

```bash
pip install pydub numpy
```

```python
import os

from pydub import AudioSegment
from pydub.silence import split_on_silence

def split_audio(file_path, output_dir="output", max_length=50, min_length=5,
                silence_thresh=-40, min_silence=600, keep_silence=200, step_size=50):
    os.makedirs(output_dir, exist_ok=True)
    audio = AudioSegment.from_file(file_path)
    raw = split_on_silence(
        audio,
        silence_thresh=silence_thresh,
        min_silence_len=min_silence,
        keep_silence=keep_silence,
        seek_step=step_size,
    )

    # 合并过短片段
    merged = []
    for chunk in raw:
        if merged and len(merged[-1]) + len(chunk) <= max_length * 1000:
            merged[-1] += chunk
        else:
            merged.append(chunk)

    # 拆分过长片段，丢弃过短尾部
    segments = []
    for chunk in merged:
        if len(chunk) > max_length * 1000:
            segments.extend(chunk[i:i + max_length * 1000]
                            for i in range(0, len(chunk), max_length * 1000))
        elif len(chunk) >= min_length * 1000:
            segments.append(chunk)

    for idx, seg in enumerate(segments, 1):
        seg.export(os.path.join(output_dir, f"segment_{idx:03d}.wav"), format="wav")
    return segments

if __name__ == "__main__":
    split_audio("1.wav", max_length=50, min_length=5)
```
