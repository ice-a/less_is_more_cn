```python
import os, shutil

path = r"文件夹路径"  # 待整理的文件夹路径
data = os.listdir(path)
count = 1
p2 = os.path.join(path,"_{}".format(count))
temp = 0
for i in data:
    p1 = os.path.join(path, i)
    if not os.path.isfile(p1):  # 跳过子目录，避免把整个目录搬走
        continue
    temp += 1
    os.makedirs(p2, exist_ok=True)  # 原 if/else 两个分支做的是同一件事，合并为一句
    shutil.move(p1, p2)
    print("move", p1, ">>>", p2)
    if temp % 5 == 0:  # 将 5 改成每个文件夹要放多少个文件
        count += 1
        p2 = os.path.join(path,"_{}".format(count))
```
